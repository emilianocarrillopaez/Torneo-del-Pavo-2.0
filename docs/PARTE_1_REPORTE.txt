1. RESUMEN EJECUTIVO
--------------------------------------------------------------------------------
Se cerraron las brechas críticas y altas de privacidad y control de acceso en
código y en las reglas de Firestore (ya desplegadas hacia la base con nombre
`ai-studio-1b982696-e484-4fe2-a9e1-54984917aae2`): los datos personales
completos (`/participantes`) y la bóveda de comprobantes (`/comprobantes`)
quedaron restringidos exclusivamente a administradores con correo verificado
vía Google Sign-In, mientras el padrón público lee únicamente la colección
sanitizada `/participantes_publicos`. Se blindaron inscripciones con lotes
atómicos (`writeBatch`), llaves SHA-256 anti-duplicados sin enumeración,
validación binaria (magic bytes) y de data URL en comprobantes, folios
impredecibles, exportaciones seguras sin `xlsx` y sin inyección de fórmulas,
Aviso de Privacidad con consentimiento obligatorio de tutor para menores de 18
años, y limpieza de datos bancarios reales del código fuente.

Riesgos que permanecen hasta completar los pasos manuales / entorno local:
falta ejecutar la suite del emulador (`npm run test:rules`) en un equipo con
Java JRE instalado, activar App Check con enforcement, desactivar
Correo/Contraseña en Firebase Auth, restringir la API key por referrer HTTP y
registrar el documento inicial del organizador en `/admins` (o
`/admins_by_email`).

--- firestore.rules ---
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Denegar todo por defecto
    match /{document=**} {
      allow read, write: if false;
    }

    function isValidId(id) {
      return id is string && id.size() > 0 && id.size() <= 128 && id.matches('^[a-zA-Z0-9_\\.@\\-]+$');
    }

    function isSignedIn() {
      return request.auth != null;
    }

    function isVerifiedUser() {
      return isSignedIn() && request.auth.token.email_verified == true;
    }

    // Sin correos hardcodeados en reglas; exige email_verified == true en todas las ramas
    function isAdmin() {
      return isVerifiedUser() && (
        exists(/databases/$(database)/documents/admins/$(request.auth.uid)) ||
        (
          request.auth.token.email is string &&
          exists(/databases/$(database)/documents/admins_by_email/$(request.auth.token.email.lower()))
        )
      );
    }

    function isRegistrationOpen() {
      return exists(/databases/$(database)/documents/configuracion/general)
        && get(/databases/$(database)/documents/configuracion/general).data.inscripcionesAbiertas == true
        && (
          !('fechaLimiteTimestamp' in get(/databases/$(database)/documents/configuracion/general).data) ||
          (
            get(/databases/$(database)/documents/configuracion/general).data.fechaLimiteTimestamp is timestamp &&
            request.time <= get(/databases/$(database)/documents/configuracion/general).data.fechaLimiteTimestamp
          )
        );
    }

    function isValidParticipant(data, isCreate) {
      return data.keys().hasOnly([
          'nombre', 'fechaNacimiento', 'whatsapp', 'correo', 'categoria',
          'estatusPago', 'fechaRegistro', 'folio', 'esFemenil', 'esAficionado',
          'tieneRecibo', 'reciboId', 'reciboNombreArchivo', 'notas', 'website',
          'aceptoAviso', 'aceptoAvisoVersion', 'consentimientoTutor', 'nombreTutor',
          'fechaConsentimiento'
        ])
        && data.keys().hasAll([
          'nombre', 'fechaNacimiento', 'whatsapp', 'correo', 'categoria',
          'estatusPago', 'fechaRegistro', 'folio', 'esFemenil', 'esAficionado'
        ])
        && data.nombre is string && data.nombre.size() >= 3 && data.nombre.size() <= 150 && data.nombre.trim() == data.nombre
        && data.fechaNacimiento is string && data.fechaNacimiento.matches('^(19[2-9][0-9]|20[0-2][0-9])-(0[1-9]|1[0-2])-(0[1-9]|[12][0-9]|3[01])$')
        && data.whatsapp is string && data.whatsapp.matches('^[0-9]{10}$')
        && data.correo is string && data.correo.size() >= 5 && data.correo.size() <= 100 && data.correo == data.correo.lower() && data.correo.matches('^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$')
        && data.categoria in ['Juvenil', 'Libre']
        && data.estatusPago in ['pendiente', 'pagado']
        && data.fechaRegistro is string && data.fechaRegistro.size() >= 10 && data.fechaRegistro.size() <= 60
        && data.folio is string && (isCreate ? data.folio.matches('^PAVO-[A-HJ-NP-Z2-9]{6}$') : data.folio.matches('^PAVO-[A-Z0-9]{4,10}$'))
        && data.esFemenil is bool
        && data.esAficionado is bool
        && (!isCreate || (
          'aceptoAviso' in data && data.aceptoAviso == true &&
          'fechaConsentimiento' in data && data.fechaConsentimiento is string && data.fechaConsentimiento.size() >= 10 && data.fechaConsentimiento.size() <= 60
        ))
        && (!('aceptoAviso' in data) || data.aceptoAviso is bool)
        && (!('fechaConsentimiento' in data) || (data.fechaConsentimiento is string && data.fechaConsentimiento.size() <= 60))
        && (!('aceptoAvisoVersion' in data) || (data.aceptoAvisoVersion is string && data.aceptoAvisoVersion.size() <= 30))
        && (!('consentimientoTutor' in data) || data.consentimientoTutor is bool)
        && (!('nombreTutor' in data) || (data.nombreTutor is string && data.nombreTutor.size() <= 150))
        && (!('tieneRecibo' in data) || data.tieneRecibo is bool)
        && (!('reciboId' in data) || (data.reciboId is string && isValidId(data.reciboId)))
        && (!('reciboNombreArchivo' in data) || (data.reciboNombreArchivo is string && data.reciboNombreArchivo.size() <= 180))
        && (!('notas' in data) || (data.notas is string && data.notas.size() <= 500))
        && (!('website' in data) || data.website == '');
    }

    function hasMinorConsentIfRequired(data) {
      return !(data.fechaNacimiento > '2008-12-19') || (
        'consentimientoTutor' in data && data.consentimientoTutor == true &&
        'nombreTutor' in data && data.nombreTutor is string && data.nombreTutor.trim().size() >= 3
      );
    }

    function isValidPublicParticipant(data) {
      return data.keys().hasOnly([
          'folio', 'nombre', 'categoria', 'esAficionado', 'esFemenil',
          'estatusPago', 'rangoEdad', 'edad', 'fechaRegistro'
        ])
        && data.keys().hasAll([
          'folio', 'nombre', 'categoria', 'esAficionado', 'esFemenil',
          'estatusPago', 'fechaRegistro'
        ])
        && data.folio is string && data.folio.matches('^PAVO-[A-Z0-9]{4,10}$')
        && data.nombre is string && data.nombre.size() >= 3 && data.nombre.size() <= 150 && data.nombre.trim() == data.nombre
        && data.categoria in ['Juvenil', 'Libre']
        && data.esAficionado is bool
        && data.esFemenil is bool
        && data.estatusPago in ['pendiente', 'pagado']
        && data.fechaRegistro is string && data.fechaRegistro.size() >= 10 && data.fechaRegistro.size() <= 60
        && (!('rangoEdad' in data) || (data.rangoEdad is string && data.rangoEdad.size() <= 40))
        && (!('edad' in data) || (data.edad is number && data.edad >= 3 && data.edad <= 110));
    }

    function isValidComprobante(data) {
      return data.keys().hasOnly([
          'participantId', 'folio', 'participantNombre', 'fileName',
          'mimeType', 'fileSize', 'imageDataUrl', 'uploadedAt'
        ])
        && data.keys().hasAll([
          'participantId', 'folio', 'participantNombre', 'fileName',
          'mimeType', 'fileSize', 'imageDataUrl', 'uploadedAt'
        ])
        && isValidId(data.participantId)
        && data.folio is string && data.folio.size() >= 3 && data.folio.size() <= 30
        && data.participantNombre is string && data.participantNombre.size() >= 3 && data.participantNombre.size() <= 150
        && data.fileName is string && data.fileName.size() >= 1 && data.fileName.size() <= 200 && !data.fileName.matches('.*[/\\\\].*')
        && data.mimeType in ['image/jpeg', 'image/png', 'image/webp', 'application/pdf']
        && data.fileSize is number && data.fileSize >= 1 && data.fileSize <= 950000
        && data.imageDataUrl is string && data.imageDataUrl.size() >= 30 && data.imageDataUrl.size() <= 950000
        && (
          (data.mimeType == 'image/jpeg' && data.imageDataUrl.matches('^data:image/jpeg;base64,[A-Za-z0-9+/=]+$')) ||
          (data.mimeType == 'image/png' && data.imageDataUrl.matches('^data:image/png;base64,[A-Za-z0-9+/=]+$')) ||
          (data.mimeType == 'image/webp' && data.imageDataUrl.matches('^data:image/webp;base64,[A-Za-z0-9+/=]+$')) ||
          (data.mimeType == 'application/pdf' && data.imageDataUrl.matches('^data:application/pdf;base64,[A-Za-z0-9+/=]+$'))
        )
        && data.uploadedAt is string && data.uploadedAt.size() <= 60;
    }

    function isValidConfig(data) {
      return data.keys().hasOnly([
          'titulo', 'edicion', 'slogan', 'club', 'fecha', 'fechaIso', 'hora',
          'lugar', 'referencia', 'mapaUrl', 'costo', 'cupoMaximo', 'totalInscritos',
          'fechaLimite', 'fechaLimiteIso', 'fechaLimiteTimestamp', 'inscripcionesAbiertas',
          'clabe', 'banco', 'titular', 'whatsappContacto', 'bolsaPremios',
          'primerLugarEfectivo', 'presidente', 'vicepresidente', 'premiacionTexto',
          'sistemaCompetenciaTexto', 'transitoriosTexto', 'indicacionesPago'
        ])
        && data.keys().hasAll(['titulo', 'fecha', 'costo', 'cupoMaximo', 'inscripcionesAbiertas'])
        && data.titulo is string && data.titulo.size() >= 3 && data.titulo.size() <= 150
        && data.fecha is string && data.fecha.size() >= 3 && data.fecha.size() <= 100
        && data.costo is number && data.costo >= 0 && data.costo <= 100000
        && data.cupoMaximo is number && data.cupoMaximo >= 1 && data.cupoMaximo <= 5000
        && data.inscripcionesAbiertas is bool
        && (!('totalInscritos' in data) || (data.totalInscritos is number && data.totalInscritos >= 0))
        && (!('mapaUrl' in data) || data.mapaUrl == '' || data.mapaUrl == '#' || data.mapaUrl.matches('^https://[^\\s]+$'))
        && (!('fechaLimiteTimestamp' in data) || data.fechaLimiteTimestamp is timestamp)
        && (!('fechaLimiteIso' in data) || (data.fechaLimiteIso is string && data.fechaLimiteIso.size() <= 60))
        && (!('whatsappContacto' in data) || (data.whatsappContacto is string && data.whatsappContacto.size() <= 20))
        && (!('clabe' in data) || (data.clabe is string && data.clabe.size() <= 30));
    }

    function isValidSponsor(data) {
      return data.keys().hasOnly(['nombre', 'logo', 'enlace', 'activo', 'orden', 'categoriaSponsor'])
        && data.keys().hasAll(['nombre', 'logo', 'activo'])
        && data.nombre is string && data.nombre.size() >= 1 && data.nombre.size() <= 100
        && data.activo is bool
        && data.logo is string && data.logo.size() <= 950000
        && (
          data.logo == '' ||
          data.logo.matches('^https://[^\\s]+$') ||
          data.logo.matches('^data:image/(jpeg|png|webp|svg\\+xml)(;base64,|;charset=UTF-8,).*$')
        )
        && (!('enlace' in data) || data.enlace == '' || data.enlace == '#' || data.enlace.matches('^https://[^\\s]+$'))
        && (!('orden' in data) || data.orden is number)
        && (!('categoriaSponsor' in data) || (data.categoriaSponsor is string && data.categoriaSponsor.size() <= 120));
    }

    function isValidAdminMember(data) {
      return data.keys().hasOnly(['email', 'nombre', 'rol', 'agregadoEn'])
        && data.keys().hasAll(['email', 'nombre', 'rol'])
        && data.email is string && data.email.size() >= 5 && data.email.size() <= 120 && data.email == data.email.lower() && data.email.matches('^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$')
        && data.nombre is string && data.nombre.size() >= 2 && data.nombre.size() <= 120
        && data.rol is string && data.rol.size() >= 2 && data.rol.size() <= 80
        && (!('agregadoEn' in data) || (data.agregadoEn is string && data.agregadoEn.size() <= 60));
    }

    // Colección privada de participantes (PII: correo, WhatsApp, nacimiento, reciboId, tutor)
    match /participantes/{participanteId} {
      allow get, list: if isAdmin();

      allow create: if isValidId(participanteId)
        && isValidParticipant(request.resource.data, true)
        && (
          isAdmin()
          || (
            isRegistrationOpen()
            && request.resource.data.estatusPago == 'pendiente'
            && request.resource.data.tieneRecibo == true
            && 'reciboId' in request.resource.data
            && existsAfter(/databases/$(database)/documents/comprobantes/$(request.resource.data.reciboId))
            && existsAfter(/databases/$(database)/documents/participantes_publicos/$(participanteId))
            && (!('notas' in request.resource.data) || request.resource.data.notas == '')
            && hasMinorConsentIfRequired(request.resource.data)
          )
        );

      allow update: if isAdmin()
        && isValidId(participanteId)
        && isValidParticipant(request.resource.data, false);

      allow delete: if isAdmin() && isValidId(participanteId);
    }

    // Colección pública sanitizada para el padrón (cero datos de contacto, nacimiento, tutor ni recibos)
    match /participantes_publicos/{participanteId} {
      allow read: if true;

      allow create: if isValidId(participanteId)
        && isValidPublicParticipant(request.resource.data)
        && existsAfter(/databases/$(database)/documents/participantes/$(participanteId))
        && (
          isAdmin()
          || (
            request.resource.data.estatusPago == 'pendiente'
            && request.resource.data.folio == getAfter(/databases/$(database)/documents/participantes/$(participanteId)).data.folio
            && request.resource.data.categoria == getAfter(/databases/$(database)/documents/participantes/$(participanteId)).data.categoria
            && request.resource.data.esAficionado == getAfter(/databases/$(database)/documents/participantes/$(participanteId)).data.esAficionado
            && request.resource.data.esFemenil == getAfter(/databases/$(database)/documents/participantes/$(participanteId)).data.esFemenil
            && request.resource.data.estatusPago == getAfter(/databases/$(database)/documents/participantes/$(participanteId)).data.estatusPago
          )
        );

      allow update: if isAdmin()
        && isValidId(participanteId)
        && isValidPublicParticipant(request.resource.data);

      allow delete: if isAdmin() && isValidId(participanteId);
    }

    // Llaves deterministas SHA-256 para evitar duplicados sin permitir enumeración
    match /registro_llaves/{hashId} {
      allow get, list, update: if false;

      allow create: if hashId is string
        && hashId.matches('^[a-f0-9]{64}$')
        && request.resource.data.keys().hasOnly(['tipo', 'participantId', 'createdAt'])
        && request.resource.data.keys().hasAll(['tipo', 'participantId', 'createdAt'])
        && request.resource.data.tipo in ['email', 'whatsapp']
        && isValidId(request.resource.data.participantId)
        && request.resource.data.createdAt is string && request.resource.data.createdAt.size() <= 60
        && existsAfter(/databases/$(database)/documents/participantes/$(request.resource.data.participantId));

      allow delete: if isAdmin() && hashId is string && hashId.matches('^[a-f0-9]{64}$');
    }

    // Bóveda privada de comprobantes (solo admin lee, actualiza o borra)
    match /comprobantes/{comprobanteId} {
      allow read: if isAdmin();

      allow create: if isValidId(comprobanteId)
        && isValidComprobante(request.resource.data)
        && (
          isAdmin()
          || (
            isRegistrationOpen()
            && existsAfter(/databases/$(database)/documents/participantes/$(request.resource.data.participantId))
          )
        );

      allow update: if isAdmin()
        && isValidId(comprobanteId)
        && isValidComprobante(request.resource.data);

      allow delete: if isAdmin() && isValidId(comprobanteId);
    }

    // Configuración del torneo (Opción A: sin escritura pública a totalInscritos)
    match /configuracion/{configId} {
      allow read: if true;
      allow create, update: if isAdmin() && isValidId(configId) && isValidConfig(request.resource.data);
      allow delete: if isAdmin() && isValidId(configId);
    }

    // Patrocinadores
    match /patrocinadores/{patrocinadorId} {
      allow read: if true;
      allow create, update: if isAdmin() && isValidId(patrocinadorId) && isValidSponsor(request.resource.data);
      allow delete: if isAdmin() && isValidId(patrocinadorId);
    }

    // Administradores por UID y por correo normalizado
    match /admins/{adminId} {
      allow get: if isVerifiedUser() && (request.auth.uid == adminId || isAdmin());
      allow list: if isAdmin();
      allow create, update: if isAdmin() && isValidId(adminId) && isValidAdminMember(request.resource.data);
      allow delete: if isAdmin() && isValidId(adminId);
    }

    match /admins_by_email/{emailId} {
      allow get: if isVerifiedUser() && (
        (request.auth.token.email is string && request.auth.token.email.lower() == emailId) ||
        isAdmin()
      );
      allow list: if isAdmin();
      allow create, update: if isAdmin()
        && isValidId(emailId)
        && emailId == emailId.lower()
        && isValidAdminMember(request.resource.data)
        && request.resource.data.email == emailId;
      allow delete: if isAdmin() && isValidId(emailId);
    }

    // Documento único de comprobación de conectividad
    match /test/{docId} {
      allow get: if docId == 'connection';
      allow list, write: if false;
    }
  }
}
